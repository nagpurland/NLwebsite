// /pages/dashboard/property-insights.js
import { useAuth } from '@/context/AuthContext';
import { getPropertyInsights } from '@/lib/api';

export default function PropertyInsights() {
  const { user } = useAuth();
  const [insights, setInsights] = useState(null);

  useEffect(() => {
    if (user?.email) {
      getPropertyInsights(user.email).then(setInsights);
    }
  }, [user]);

  if (!insights) return <Loading />;

  return (
    <div className="space-y-6">
      <ShortlistSummary data={insights.summary} />
      <MarketInsights data={insights.market} />
      <InvestmentScoreCalculator />
    </div>
  );
}
